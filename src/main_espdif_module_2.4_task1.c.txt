#include <stdio.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/gpio.h"
#include "esp_log.h"
#include "esp_attr.h"

static const char *TAG = "button_bounce";

#define BUTTON_GPIO  GPIO_NUM_16   // змініть за потреби

static volatile uint32_t press_count = 0;
static uint32_t last_printed = 0;

// ISR: тільки інкремент
static void IRAM_ATTR button_isr_handler(void *arg)
{
    press_count++;
}

void app_main(void)
{
    // Налаштування GPIO
    gpio_config_t io_conf = {
        .pin_bit_mask = (1ULL << BUTTON_GPIO),
        .mode = GPIO_MODE_INPUT,
        .pull_up_en = GPIO_PULLUP_ENABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_NEGEDGE   // FALLING
    };
    gpio_config(&io_conf);

    // Сервіс переривань + обробник
    gpio_install_isr_service(0);
    gpio_isr_handler_add(BUTTON_GPIO, button_isr_handler, NULL);

    ESP_LOGI(TAG, "Button bounce demo started. Press the button...");

    while (1) {
        if (press_count != last_printed) {
            last_printed = press_count;
            ESP_LOGI(TAG, "Interrupt count: %lu", (unsigned long)last_printed);
        }
        vTaskDelay(pdMS_TO_TICKS(10));
    }
}
