#include <stdio.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/gpio.h"
#include "esp_log.h"
#include "esp_attr.h"

static const char *TAG = "debounce_state";

#define BUTTON_GPIO  GPIO_NUM_16

static volatile bool button_event = false;  // ISR лише сигналізує подію
static bool press_handled = false;
static uint32_t press_count = 0;

static void IRAM_ATTR button_isr_handler(void *arg)
{
    button_event = true;
}

void app_main(void)
{
    gpio_config_t io_conf = {
        .pin_bit_mask  = (1ULL << BUTTON_GPIO),
        .mode          = GPIO_MODE_INPUT,
        .pull_up_en    = GPIO_PULLUP_ENABLE,
        .pull_down_en  = GPIO_PULLDOWN_DISABLE,
        .intr_type     = GPIO_INTR_NEGEDGE  // FALLING
    };
    gpio_config(&io_conf);

    gpio_install_isr_service(0);
    gpio_isr_handler_add(BUTTON_GPIO, button_isr_handler, NULL);

    ESP_LOGI(TAG, "State-based debounce. Press the button...");

    while (1) {
        if (button_event) {
            button_event = false;

            // прийняти лише якщо кнопка досі натиснута і ще не обробляли
            if (gpio_get_level(BUTTON_GPIO) == 0 && !press_handled) {
                press_handled = true;
                press_count++;
                ESP_LOGI(TAG, "Valid press: %lu", (unsigned long)press_count);
            }
            // HIGH = відпускання / брязкіт → ігноруємо
        }

        // відпустили кнопку → дозволити наступну реакцію
        if (gpio_get_level(BUTTON_GPIO) == 1) {
            press_handled = false;
        }

        vTaskDelay(pdMS_TO_TICKS(10));  // щоб не ловити task_wdt
    }
}
