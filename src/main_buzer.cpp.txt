#include <Arduino.h>

const int ldrPin = 6;  
const int ledPin = 5;  
int lightValue = 0;

const int threshold = 2600; 

void setup() {
  Serial.begin(115200);
  pinMode(ledPin, OUTPUT);
}

void loop() {
  lightValue = analogRead(ldrPin);
  
  Serial.print("Level: ");
  Serial.println(lightValue);


  if (lightValue < threshold) {
    digitalWrite(ledPin, HIGH); 
  } else {
    digitalWrite(ledPin, LOW);  
  }
  
  delay(200); 
}
