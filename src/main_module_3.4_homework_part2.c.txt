#include <stdio.h>
#include "driver/ledc.h"
#include "driver/gpio.h"
#include "esp_timer.h"
#include "esp_log.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

static const char *TAG = "buzzer_melody";

#define BUZZER_GPIO   GPIO_NUM_2
#define LEDC_MODE     LEDC_LOW_SPEED_MODE
#define LEDC_TIMER    LEDC_TIMER_0
#define LEDC_CH       LEDC_CHANNEL_0
#define LEDC_RES      LEDC_TIMER_10_BIT
#define DUTY_ON       512
#define TICK_US       (50 * 1000)

/* Ноти (Гц) */
#define REST     0
#define NOTE_C4  262
#define NOTE_D4  294
#define NOTE_E4  330

typedef struct {
    uint16_t freq;
    uint16_t dur_ms;
} tone_t;


// We Will Rock You (спрощено з умови):

static const tone_t melody[] = {
    {NOTE_C4, 200}, {NOTE_C4, 200}, {NOTE_D4, 400},
    {NOTE_C4, 200}, {NOTE_C4, 200}, {NOTE_D4, 400},
    {NOTE_C4, 200}, {NOTE_C4, 200}, {NOTE_D4, 200}, {NOTE_E4, 600},
    {REST, 800}
};

static const size_t melody_len = sizeof(melody) / sizeof(melody[0]);

static size_t  idx = 0;
static int64_t elapsed_ms = 0;

static void ledc_buzzer_init(void)
{
    ledc_timer_config_t tcfg = {
        .speed_mode      = LEDC_MODE,
        .duty_resolution = LEDC_RES,
        .timer_num       = LEDC_TIMER,
        .freq_hz         = 1000,
        .clk_cfg         = LEDC_AUTO_CLK
    };
    ESP_ERROR_CHECK(ledc_timer_config(&tcfg));

    ledc_channel_config_t ccfg = {
        .gpio_num   = BUZZER_GPIO,
        .speed_mode = LEDC_MODE,
        .channel    = LEDC_CH,
        .timer_sel  = LEDC_TIMER,
        .duty       = 0,
        .hpoint     = 0
    };
    ESP_ERROR_CHECK(ledc_channel_config(&ccfg));
}

static void set_tone(uint16_t freq_hz)
{
    if (freq_hz == 0) {
        ledc_set_duty(LEDC_MODE, LEDC_CH, 0);
        ledc_update_duty(LEDC_MODE, LEDC_CH);
        return;
    }
    ledc_set_freq(LEDC_MODE, LEDC_TIMER, freq_hz);
    ledc_set_duty(LEDC_MODE, LEDC_CH, DUTY_ON);
    ledc_update_duty(LEDC_MODE, LEDC_CH);
}

static void tick_cb(void *arg)
{
    elapsed_ms += 50;

    if (elapsed_ms >= melody[idx].dur_ms) {
        elapsed_ms = 0;
        idx++;
        if (idx >= melody_len) {
            idx = 0;
        }
        set_tone(melody[idx].freq);
    }
}

void app_main(void)
{
    ledc_buzzer_init();
    set_tone(melody[0].freq);

    const esp_timer_create_args_t args = {
        .callback = &tick_cb,
        .arg = NULL,
        .dispatch_method = ESP_TIMER_TASK,
        .name = "bz_tick"
    };
    esp_timer_handle_t tmr;
    ESP_ERROR_CHECK(esp_timer_create(&args, &tmr));
    ESP_ERROR_CHECK(esp_timer_start_periodic(tmr, TICK_US));

    ESP_LOGI(TAG, "Module 3.4 Task 2: We Will Rock You (simplified)");

    while (1) {
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}
